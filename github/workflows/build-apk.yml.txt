name: Build Android APK

# Runs entirely on GitHub's cloud servers — no PC, no Android Studio
# needed on your end. Trigger it manually from the "Actions" tab
# (works from the GitHub mobile site or the GitHub mobile app), or it
# runs automatically on every push to main.
#
# This workflow expects "mandal-app.zip" (the full project archive) to
# be committed at the repo root AS-IS — no manual extraction needed.
# Mobile browsers can't upload a nested folder structure reliably, but
# uploading one single .zip file works fine, so this workflow unzips
# it itself as the first build step.
on:
  workflow_dispatch:
  push:
    branches: [main]

jobs:
  build-apk:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Extract project archive
        run: unzip -o mandal-app.zip

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: 20

      - name: Setup Java (required by Android Gradle build)
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17

      - name: Install frontend dependencies
        working-directory: mandal-app/frontend
        run: npm install

      - name: Create production environment file
        working-directory: mandal-app/frontend
        # VITE_API_BASE_URL must be set as a GitHub Actions secret
        # (Settings → Secrets and variables → Actions → New repository
        # secret) before running this workflow — see the repo README.
        run: echo "VITE_API_BASE_URL=${{ secrets.VITE_API_BASE_URL }}" > .env.production

      - name: Build frontend
        working-directory: mandal-app/frontend
        run: npm run build

      - name: Add Android platform (first time only — safe to re-run)
        working-directory: mandal-app/frontend
        run: npx cap add android

      - name: Sync web build into Android project
        working-directory: mandal-app/frontend
        run: npx cap sync android

      - name: Build debug APK
        working-directory: mandal-app/frontend/android
        run: |
          chmod +x gradlew
          ./gradlew assembleDebug

      - name: Attach APK to a GitHub Release
        uses: softprops/action-gh-release@v2
        with:
          tag_name: build-${{ github.run_number }}
          name: Mandal App build #${{ github.run_number }}
          files: mandal-app/frontend/android/app/build/outputs/apk/debug/app-debug.apk
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
